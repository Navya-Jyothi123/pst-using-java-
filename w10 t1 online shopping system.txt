
import java.util.*;

class Product {
    int id;
    String name;
    double price;

    Product(int id, String name, double price) {
        this.id = id;
        this.name = name;
        this.price = price;
    }

    void display() {
        System.out.println("ID: " + id +
                ", Name: " + name +
                ", Price: Rs." + price);
    }
}

class Order {
    int orderId;
    int priority;

    Order(int orderId, int priority) {
        this.orderId = orderId;
        this.priority = priority;
    }
}

public class OnlineShoppingSystem {

    public static void main(String[] args) {

        // Linear collection: ArrayList
        ArrayList<Product> products = new ArrayList<>();

        // Non-linear collection: HashMap
        HashMap<Integer, Product> productMap = new HashMap<>();

        // Non-linear collection: PriorityQueue (max-heap)
        PriorityQueue<Order> orders = new PriorityQueue<>(
            (a, b) -> Integer.compare(b.priority, a.priority)
        );

        // Add products
        Product p1 = new Product(101, "Laptop", 50000);
        Product p2 = new Product(102, "Mobile", 20000);
        Product p3 = new Product(103, "Headphones", 2000);

        products.add(p1);
        products.add(p2);
        products.add(p3);

        productMap.put(p1.id, p1);
        productMap.put(p2.id, p2);
        productMap.put(p3.id, p3);

        // Display all products
        System.out.println("Product List:");
        for (Product p : products) {
            p.display();
        }

        // Search product by ID
        int searchId = 102;
        System.out.println("\nSearching Product ID: " + searchId);

        Product found = productMap.get(searchId);

        if (found != null) {
            found.display();
        } else {
            System.out.println("Product not found.");
        }

        // Add orders with priorities
        orders.add(new Order(201, 2));
        orders.add(new Order(202, 5));
        orders.add(new Order(203, 3));

        // Process orders by priority
        System.out.println("\nProcessing Orders:");

        while (!orders.isEmpty()) {
            Order o = orders.poll();
            System.out.println("Order ID: " + o.orderId +
                    ", Priority: " + o.priority);
        }
    }
}
