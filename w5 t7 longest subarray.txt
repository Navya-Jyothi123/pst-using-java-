class Solution {
    public int lengthOfLongestSubstring(String s) {
        int left = 0;
        int max = 0;

        for (int right = 0; right < s.length(); right++) {

            while (s.substring(left, right).contains(
                    String.valueOf(s.charAt(right)))) {
                left++;
            }

            max = Math.max(max, right - left + 1);
        }

        return max;
    }
}
