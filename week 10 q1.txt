
import java.util.*;

class Product {
    int id;
    String name;
    double price;

    Product(int id, String name, double price) {
        this.id = id;
        this.name = name;
        this.price = price;
    }

    public String toString() {
        return id + " " + name + " " + price;
    }
}

class Order {
    int orderId;
    int priority;

    Order(int orderId, int priority) {
        this.orderId = orderId;
        this.priority = priority;
    }
}

public class Main {
    public static void main(String[] args) {

        // Linear collection
        ArrayList<Product> products = new ArrayList<>();

        products.add(new Product(101, "Laptop", 50000));
        products.add(new Product(102, "Mobile", 20000));
        products.add(new Product(103, "Tablet", 15000));

        System.out.println("Product List:");
        for (Product p : products) {
            System.out.println(p);
        }

        // Fast searching using HashMap
        HashMap<Integer, Product> map = new HashMap<>();

        for (Product p : products) {
            map.put(p.id, p);
        }

        System.out.println("\nSearch Product ID 102:");
        System.out.println(map.get(102));

        // Priority queue for orders
        PriorityQueue<Order> orders =
            new PriorityQueue<>(
                (a, b) -> a.priority - b.priority
            );

        orders.add(new Order(1, 3));
        orders.add(new Order(2, 1));
        orders.add(new Order(3, 2));

        System.out.println("\nOrders by Priority:");
        while (!orders.isEmpty()) {
            Order o = orders.poll();
            System.out.println(
                "Order ID: " + o.orderId +
                " Priority: " + o.priority
            );
        }
    }
}
