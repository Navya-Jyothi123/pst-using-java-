class Solution {
    public int[][] matrixBlockSum(int[][] mat, int k) {
        int m = mat.length;
        int n = mat[0].length;

        int[][] sum = new int[m + 1][n + 1];
        int[][] answer = new int[m][n];

        // Prefix sum
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                sum[i + 1][j + 1] = mat[i][j]
                        + sum[i][j + 1]
                        + sum[i + 1][j]
                        - sum[i][j];
            }
        }

        // Find block sum
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                int r1 = Math.max(0, i - k);
                int r2 = Math.min(m - 1, i + k);
                int c1 = Math.max(0, j - k);
                int c2 = Math.min(n - 1, j + k);

                answer[i][j] = sum[r2 + 1][c2 + 1]
                        - sum[r1][c2 + 1]
                        - sum[r2 + 1][c1]
                        + sum[r1][c1];
            }
        }

        return answer;
    }
}
