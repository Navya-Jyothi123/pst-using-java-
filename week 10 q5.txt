
import java.util.*;

class Node {
    String song;
    Node next;

    Node(String song) {
        this.song = song;
        this.next = null;
    }
}

public class Main {

    Node head = null;

    // Add song
    void addSong(String song) {
        Node newNode = new Node(song);

        if (head == null) {
            head = newNode;
        } else {
            Node temp = head;

            while (temp.next != null) {
                temp = temp.next;
            }

            temp.next = newNode;
        }
    }

    // Remove song
    void removeSong(String song) {
        if (head == null) {
            System.out.println("Playlist is empty");
            return;
        }

        if (head.song.equals(song)) {
            head = head.next;
            return;
        }

        Node temp = head;

        while (temp.next != null &&
               !temp.next.song.equals(song)) {
            temp = temp.next;
        }

        if (temp.next != null) {
            temp.next = temp.next.next;
        } else {
            System.out.println("Song not found");
        }
    }

    // Display playlist
    void display() {
        Node temp = head;

        System.out.println("Music Playlist:");

        while (temp != null) {
            System.out.println(temp.song);
            temp = temp.next;
        }
    }

    public static void main(String[] args) {

        Main playlist = new Main();

        playlist.addSong("Shape of You");
        playlist.addSong("Perfect");
        playlist.addSong("Believer");
        playlist.addSong("Memories");

        playlist.display();

        System.out.println("\nAfter removing a song:");

        playlist.removeSong("Believer");

        playlist.display();
    }
}
