
import java.io.*;
import java.util.*;

public class Solution {

    static int[] palindrome(char[] str) {
        int n = str.length;
        int[] r = new int[2 * n];
        int k = 0;

        for (int i = 0, j = 0; i < 2 * n; i += k,
             j = Math.max(j - k, 0)) {

            while (i - j >= 0 && i + j + 1 < 2 * n
                    && str[(i - j) / 2] == str[(i + j + 1) / 2]) {
                j++;
            }

            r[i] = j;

            for (k = 1; i - k >= 0 && r[i] - k >= 0
                    && r[i - k] != r[i] - k; k++) {
                r[i + k] = Math.min(r[i - k], r[i] - k);
            }
        }

        return r;
    }

    static class MaxHeap {
        int[] a, map, imap;
        int pos = 1;

        MaxHeap(int n) {
            a = new int[n + 2];
            map = new int[n + 2];
            imap = new int[n + 2];

            Arrays.fill(a, Integer.MIN_VALUE);
            Arrays.fill(map, -1);
            Arrays.fill(imap, -1);
        }

        int size() {
            return pos - 1;
        }

        int max() {
            return a[1];
        }

        void add(int ind, int x) {
            if (imap[ind] >= 0) return;

            a[pos] = x;
            map[pos] = ind;
            imap[ind] = pos;
            up(pos++);
        }

        void remove(int ind) {
            if (imap[ind] < 0) return;

            int rem = imap[ind];
            pos--;

            if (rem != pos) {
                a[rem] = a[pos];
                map[rem] = map[pos];
                imap[map[rem]] = rem;
            }

            a[pos] = Integer.MIN_VALUE;
            map[pos] = -1;
            imap[ind] = -1;

            if (rem < pos) {
                up(rem);
                down(rem);
            }
        }

        void up(int i) {
            while (i > 1 && a[i] > a[i / 2]) {
                swap(i, i / 2);
                i /= 2;
            }
        }

        void down(int i) {
            while (2 * i < pos) {
                int c = 2 * i;

                if (c + 1 < pos && a[c + 1] > a[c]) {
                    c++;
                }

                if (a[c] <= a[i]) break;

                swap(i, c);
                i = c;
            }
        }

        void swap(int i, int j) {
            int t = a[i];
            a[i] = a[j];
            a[j] = t;

            t = map[i];
            map[i] = map[j];
            map[j] = t;

            imap[map[i]] = i;
            imap[map[j]] = j;
        }
    }

    static void circularPalindromes(String s) {
        int n = s.length();

        char[] doubled = new char[2 * n];

        for (int i = 0; i < n; i++) {
            doubled[i] = s.charAt(i);
            doubled[i + n] = s.charAt(i);
        }

        int[] pal = palindrome(doubled);

        long[] events = new long[16 * n];
        int p = 0;

        for (int i = 0; i < 4 * n; i += 2) {
            pal[i] = Math.min(pal[i], n - ((n & 1) ^ 1));

            events[p++] = ((long) (i / 2) << 32) | i;
            events[p++] = ((long) (i / 2 + pal[i] / 2) << 32) | i;
            events[p++] = ((long) (i / 2 + n - pal[i] / 2 - 1) << 32) | i;
            events[p++] = ((long) (i / 2 + n) << 32) | i;
        }

        for (int i = 1; i < 4 * n; i += 2) {
            pal[i] = Math.min(pal[i], n - (n & 1));

            events[p++] = ((long) (i / 2) << 32) | i;
            events[p++] = ((long) (i / 2 + pal[i] / 2) << 32) | i;
            events[p++] = ((long) (i / 2 + n - pal[i] / 2) << 32) | i;
            events[p++] = ((long) (i / 2 + n) << 32) | i;
        }

        Arrays.sort(events, 0, p);

        MaxHeap inc = new MaxHeap(4 * n + 1);
        MaxHeap dec = new MaxHeap(4 * n + 1);
        MaxHeap flat = new MaxHeap(4 * n + 1);

        int[] state = new int[4 * n];
        int q = 0;

        StringBuilder out = new StringBuilder();

        for (int i = 0; i < 2 * n - 1; i++) {
            while (q < p && (events[q] >>> 32) <= i) {
                int ind = (int) events[q];

                if (state[ind] == 0) {
                    inc.add(ind, (pal[ind] & 1) - 2 * i);
                } else if (state[ind] == 1) {
                    inc.remove(ind);
                    flat.add(ind, pal[ind]);
                } else if (state[ind] == 2) {
                    flat.remove(ind);
                    dec.add(ind, pal[ind] + 2 * i);
                } else if (state[ind] == 3) {
                    dec.remove(ind);
                }

                state[ind]++;
                q++;
            }

            if (i >= n - 1) {
                int max = 0;

                if (inc.size() > 0) {
                    max = Math.max(max, inc.max() + 2 * i);
                }

                if (dec.size() > 0) {
                    max = Math.max(max, dec.max() - 2 * i);
                }

                if (flat.size() > 0) {
                    max = Math.max(max, flat.max());
                }

                out.append(max).append('\n');
            }
        }

        System.out.print(out);
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(
            new InputStreamReader(System.in)
        );

        int n = Integer.parseInt(br.readLine().trim());
        String s = br.readLine().trim();

        circularPalindromes(s);
    }
}
