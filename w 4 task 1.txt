class Solution {
    public boolean halvesAreAlike(String s) {
        int count1 = 0;
        int count2 = 0;
        int n = s.length() / 2;

        for (int i = 0; i < n; i++) {
            if ("aeiouAEIOU".indexOf(s.charAt(i)) != -1)
                count1++;
        }

        for (int i = n; i < s.length(); i++) {
            if ("aeiouAEIOU".indexOf(s.charAt(i)) != -1)
                count2++;
        }

        return count1 == count2;
    }
}
